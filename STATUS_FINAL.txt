  └── {user_id}
      ├── nome: string
      ├── email: string
      ├── senha_hash: string
      ├── data_registro: timestamp
      └── foto_perfil_url: url

  └── {user_id}
      └── {data: YYYY-MM-DD}
          ├── rotina: text
          ├── journal: text
          ├── emocoes: array
          ├── habilidades: array
          ├── checklist: array
          │   └── {texto, feito}
          ├── photo_url: url
          └── timestamp: number
